% --- Brand palette ---------------------------------------------------------
% Four colours, and each has one job. The hues run 54 -> 0 -> 351 -> 335
% degrees and the lightness falls 63 -> 69 -> 47 -> 23: one sweep from yellow
% through coral into red, then into the cover's deep maroon.
%
%   calangoyellow  ACCENT ONLY -- 1.2:1 on white, so it is used for rules,
%                  bands and marks on dark grounds, never for text on paper.
%   calangocoral   the mid tone. 3.0:1 on white: still an accent, not a text
%                  colour, but strong enough for a heavy rule or a callout's
%                  left bar to read at a glance.
%   calangored     the anchor: 5.2:1 on white. Headings' numbers, links,
%                  callout titles -- anything that has to be read.
%   calangocover   COVER GROUND ONLY. Named for the restriction: it is the one
%                  colour with a single permitted use, and a name that says so
%                  is harder to misuse than a comment. 24 points of lightness
%                  below calangored -- a visible step rather than a shade of
%                  the same thing -- and still red at hue 335 rather than
%                  collapsing to black. White sits on it at 11.6:1 and the
%                  yellow at 9.5:1, which is what the cover is set in.
\definecolor{calangoyellow}{HTML}{FEEC41}
\definecolor{calangocoral}{HTML}{FF6060}
\definecolor{calangored}{HTML}{D51C39}
\definecolor{calangocover}{HTML}{280010} % 760031, 4F0020, 280010
% Neutrals. Not part of the four: paper, ink and the greys between them.
% calangodark is the logo mark's own near-black, so body headings and the
% logo sit on the same value.
\definecolor{calangodark}{RGB}{26,26,26}
\definecolor{codebg}{RGB}{250,246,247}
\definecolor{shadegray}{RGB}{96,92,94}
\definecolor{panelgray}{RGB}{240,233,235}

% points into the four above.
\colorlet{calangoteal}{calangored}
\colorlet{tipgreen}{calangocoral}
\colorlet{warnamber}{calangoyellow}

% --- Callout boxes ---------------------------------------------------------
% The three kinds are told apart by the colour of the left rule -- red,
% coral, yellow -- and never by the title, which stays calangored in all
% three because it is the only one of the four legible as small bold text.
\newtcolorbox{calnote}[1][]{%
  breakable, enhanced, colback=calangored!5, colframe=calangored,
  boxrule=0pt, leftrule=3pt, arc=1pt, left=8pt, right=8pt, top=6pt, bottom=6pt,
  fonttitle=\bfseries\small, title={Note}, coltitle=calangored,
  attach title to upper=\par\vspace{2pt}, #1}
\newtcolorbox{caltip}[1][]{%
  breakable, enhanced, colback=calangocoral!8, colframe=calangocoral,
  boxrule=0pt, leftrule=3pt, arc=1pt, left=8pt, right=8pt, top=6pt, bottom=6pt,
  fonttitle=\bfseries\small, title={Tip}, coltitle=calangored,
  attach title to upper=\par\vspace{2pt}, #1}
